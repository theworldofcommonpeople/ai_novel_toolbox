项目结构

lib/ ├── main.dart # 应用入口 ├── models/ │ ├── api_config.dart # API配置模型 │ ├── character_card.dart # 角色卡模型 │ ├── entity_relation.dart # 实体关系模型 │ ├── generation_history.dart # 历史记录模型 │ └── novel_project.dart # 小说项目模型(含ChapterModel) ├── services/ │ ├── ai_service.dart # 统一AI调用接口(云端优先+本地回退) │ ├── cloud_api_service.dart # OpenAI兼容API(SSE流式+非流式+图像) │ ├── entity_extractor.dart # 实体抽取(人名/地名/新角色检测) │ ├── image_gen_service.dart # 封面图片生成与保存 │ ├── local_model_service.dart # 本地GGUF模型推理 │ ├── rag_service.dart # RAG检索增强(TF-IDF向量化) │ └── storage_service.dart # SQLite数据库服务(8张表) ├── pages/ │ ├── home_page.dart # 主页2列网格(16个功能卡片) │ ├── generator_base.dart # 12个文本生成器的复用基页 │ ├── bookname_test_page.dart # 书名叠加封面模板预览 │ ├── cover_generator_page.dart # AI封面生成 │ ├── consistency_check_page.dart # 一致性检查+逐章修改+Diff │ ├── character_list_page.dart # 角色卡管理(CRUD+展开详情) │ └── settings_page.dart # 5区设置(云API/本地模型/RAG) ├── widgets/ │ ├── parameter_form.dart # 动态参数表单(text/dropdown/number/multiline) │ ├── result_display.dart # AI结果显示(加载/空/内容三态) │ ├── history_list.dart # 历史记录(筛选/详情/滑动删除) │ └── diff_viewer.dart # 文本差异对比(LCS算法,并排/内联) └── utils/ ├── constants.dart # 12个生成器元数据+颜色方案+封面模板+姓氏库 ├── prompt_templates.dart # 12个生成器的系统提示词与参数定义 └── text_splitter.dart # 文本分块(自然断点优化)+中文字数统计
